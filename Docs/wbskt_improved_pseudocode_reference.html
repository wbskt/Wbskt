// Immutable, versioned, stored in SQL
record WorkflowDefinition {
    Id:              Guid          // DB primary key
    WorkspaceId:     Guid          // tenant scope — NOT the workflow id
    Name:            string
    Version:         int           // bumped on publish, old instances keep old version
    IsEnabled:       bool

    Nodes:           List<NodeDefinition>
    Edges:           List<EdgeDefinition>
    // edge = { SourceNodeId, SourcePort, TargetNodeId, TargetPort }

    InitialState:    Dictionary<string, JsonElement>
    // JsonElement = deep-cloned on instance creation (fixes shallow copy bug)

    ConcurrencyPolicy: ConcurrencyPolicy
    // AllowParallel | Queue | CancelPrevious | DropNew

    ErrorPolicy:     ErrorPolicy
    // { MaxRetries, BackoffMs, OnExhausted: Fail|Compensate|Skip }
}

record NodeDefinition {
    NodeId:      string          // stable across versions e.g. "check-temp-gate"
    TypeKey:     string          // keyed DI key e.g. "LogicGate", "DeviceTrigger"
    Config:      JsonElement    // deep-cloned per execution
    RetryPolicy: RetryPolicy?   // node-level override of definition policy
    Timeout:     TimeSpan?       // per-node execution deadline
}

record EdgeDefinition {
    SourceNodeId: string
    SourcePort:   string   // e.g. "true", "false", "done", "timeout"
    TargetNodeId: string
    TargetPort:   string
}

WorkflowRuntimeRegistry — trigger routing concurrency enforced
class WorkflowRuntimeRegistry : IHostedService {

    // Loaded once at startup, refreshed on definition change events
    index: ConcurrentDictionary<string, List<WorkflowDefinition>>
    // key = trigger key pattern e.g. "device:{deviceId}:motion"

    function OnTriggerReceived(event: DeviceEvent):
        key = BuildTriggerKey(event)          // "device:7a3b:motion"
        definitions = index[key] ?? []

        for each def in definitions where def.IsEnabled:
            // --- concurrency policy enforcement (was ignored before) ---
            existing = store.GetRunningInstances(def.Id, event.CorrelationId)

            switch def.ConcurrencyPolicy:
                case AllowParallel:
                    engine.StartAsync(def, event)

                case Queue:
                    if existing.Count == 0:
                        engine.StartAsync(def, event)
                    else:
                        queue.Enqueue(def.Id, event)  // dequeue when instance completes

                case CancelPrevious:
                    for each inst in existing:
                        engine.CancelAsync(inst, reason: "superseded")
                    engine.StartAsync(def, event)

                case DropNew:
                    if existing.Count > 0: return
                    engine.StartAsync(def, event)
}
StartAsync — instance creation
function StartAsync(def: WorkflowDefinition, event: DeviceEvent):

    // Deep clone initial state — fixes shallow copy corruption
    initialState = DeepClone(def.InitialState)

    instance = new WorkflowInstance {
        InstanceId:    Guid.NewGuid()
        DefinitionId:  def.Id           // DB PK, not WorkspaceId
        Version:       def.Version
        WorkspaceId:   def.WorkspaceId
        CorrelationId: event.CorrelationId
        Status:        Running
        State:         { Variables: initialState, NodeOutputs: {}, ExecutionLog: [] }
        Pointers:      []
    }

    // Find the trigger node matching this event
    triggerNode = def.Nodes
        .Where(n => n.TypeKey == "DeviceTrigger")
        .FirstOrDefault(n => TriggerMatches(n, event))

    if triggerNode == null: throw NoMatchingTrigger

    pointer = new ExecutionPointer {
        PointerId:     Guid.NewGuid()
        CurrentNodeId: triggerNode.NodeId
        Status:        Active
    }
    instance.Pointers.Add(pointer)

    // Persist BEFORE executing — crash safety
    await store.SaveInstanceAsync(instance)

    await ExecutePointerAsync(instance, pointer, def, triggerContext: event)
Node execution loop — the inner engine
function ExecutePointerAsync(
        instance: WorkflowInstance,
        pointer:  ExecutionPointer,
        def:      WorkflowDefinition,
        triggerCtx: TriggerContext?):

    while pointer.Status == Active:

        node = def.Nodes.Get(pointer.CurrentNodeId)

        // Scoped DI per node — fixes captive dependency
        using scope = services.CreateScope():
            executor = scope.GetKeyedService<INodeExecutor>(node.TypeKey)

            ctx = new ExecutionContext {
                InstanceId:   instance.InstanceId
                NodeId:       node.NodeId
                Config:       DeepClone(node.Config)
                TriggerCtx:   triggerCtx
                State:        instance.State       // shared ref, write via SetOutput
                Services:     scope
            }

            logEntry = { NodeId: node.NodeId, StartedAt: Now, Attempt: pointer.AttemptCount }

            result = await ExecuteWithRetryAsync(executor, ctx, node.RetryPolicy)

            logEntry.CompletedAt = Now
            logEntry.Outcome = result.ActivatedPort
            instance.State.ExecutionLog.Add(logEntry)

            if result.IsError:
                await HandleErrorAsync(instance, pointer, result.Error, def)
                return

            if result.IsSuspend:
                await SuspendPointerAsync(instance, pointer, result.Bookmark)
                return   // exit loop — pointer is parked

            // Store this node's output under its own key
            if result.Output != null:
                instance.State.NodeOutputs[node.NodeId] = DeepClone(result.Output)

            // Navigate the graph
            outboundEdges = def.Edges
                .Where(e => e.SourceNodeId == node.NodeId
                          && e.SourcePort == result.ActivatedPort)

            if outboundEdges.Count == 0:
                pointer.Status = Completed   // terminal node
                break

            if outboundEdges.Count == 1:
                pointer.CurrentNodeId = outboundEdges[0].TargetNodeId
                // continue while loop — same pointer, next node

            else:
                // Fan-out: multiple edges from one port
                await FanOutAsync(instance, pointer, outboundEdges, def)
                return

        await store.SaveInstanceAsync(instance)  // checkpoint after each node

    CheckWorkflowCompletion(instance)
Fan-out — parallel branch spawning
function FanOutAsync(
        instance: WorkflowInstance,
        sourcePointer: ExecutionPointer,
        edges: List<EdgeDefinition>,
        def: WorkflowDefinition):

    // Retire the source pointer
    sourcePointer.Status = Completed

    // Find the join node — all edges must eventually converge (optional)
    joinNodeId = FindJoinNode(edges, def)  // null if branches are independent

    if joinNodeId != null:
        // Create fan-in gate BEFORE spawning branches
        gate = new FanInGate {
            GateId:           joinNodeId
            InstanceId:       instance.InstanceId
            ExpectedBranches: edges.Count
            ArrivedBranches:  0
        }
        await store.SaveGateAsync(gate)

    // Spawn one pointer per outbound edge — fire and forget onto task scheduler
    tasks = edges.Select(edge => {
        branchPointer = new ExecutionPointer {
            PointerId:     Guid.NewGuid()
            CurrentNodeId: edge.TargetNodeId
            Status:        Active
            FanInGateId:   joinNodeId
        }
        instance.Pointers.Add(branchPointer)

        // Each branch runs on the thread pool independently
        return Task.Run(() =>
            ExecutePointerAsync(instance, branchPointer, def, triggerCtx: null))
    })

    await store.SaveInstanceAsync(instance)

    // Branches run concurrently; we do NOT await them here.
    // Each branch will call CheckFanInGate when it reaches the join node.
    _ = Task.WhenAll(tasks)  // fire-and-observe, errors surfaced via HandleError

// Called when a branch pointer arrives at a node that has a gate
function CheckFanInGateAsync(
        instance: WorkflowInstance,
        pointer:  ExecutionPointer,
        nodeId:   string) -> bool:  // true = this branch may proceed

    gate = store.GetGate(instance.InstanceId, nodeId)
    if gate == null: return true  // not a join node

    // Atomic increment via Redis INCR — safe under concurrent branch arrivals
    arrived = await store.IncrementGateAsync(gate.GateId, instance.InstanceId)

    if arrived < gate.ExpectedBranches:
        // Not all branches done yet — this branch parks itself
        pointer.Status = Suspended
        pointer.SuspendedOn = "fanin:{gate.GateId}"
        await store.SaveInstanceAsync(instance)
        return false

    // Last branch to arrive — it wins and continues past the join node
    // All other parked branch pointers are retired
    await store.RetireParkedBranchPointersAsync(instance.InstanceId, gate.GateId)
    await store.DeleteGateAsync(gate.GateId, instance.InstanceId)
    return true   // winning branch proceeds into the join node's executor

Suspend + persist — crash-safe parking
function SuspendPointerAsync(
        instance: WorkflowInstance,
        pointer:  ExecutionPointer,
        bm:       BookmarkRequest):

    // BookmarkRequest comes from the node executor:
    // { Name, Payload, TTL? }

    bookmark = new Bookmark {
        BookmarkId:     Guid.NewGuid()
        Name:           bm.Name              // e.g. "device:7a3b:temperature"
        InstanceId:     instance.InstanceId
        PointerId:      pointer.PointerId
        ResumeAtNodeId: pointer.CurrentNodeId
        CorrelationId:  instance.CorrelationId
        Payload:        bm.Payload
        ExpiresAt:      bm.TTL != null ? Now + bm.TTL : null
    }

    pointer.Status = Suspended
    pointer.SuspendedOn = bm.Name
    instance.Status = AllPointersSuspendedOrDone(instance) ? Suspended : Running

    // Transactional write — both or neither
    await store.SaveBookmarkAndInstanceAsync(bookmark, instance)
    // Process can now die. Bookmark survives in Redis.

    // Special case: delay node registers a timed resume
    if bm.IsDelay:
        delayed = new DelayedResume {
            InstanceId:   instance.InstanceId
            PointerId:    pointer.PointerId
            ResumeAt:     Now + bm.Delay
            BookmarkName: bm.Name
        }
        await store.SaveDelayedResumeAsync(delayed)
        // Background scheduler publishes resume event at ResumeAt time
Resume from event — bookmark lookup + re-entry
// Called by MassTransit consumer when a device event arrives
function ResumeAsync(event: DeviceEvent):

    bookmarkName = BuildBookmarkName(event)
    // e.g. "device:7a3b:temperature"

    // Find ALL matching bookmarks — one event may resume multiple workflows
    bookmarks = await store.FindBookmarksAsync(bookmarkName)

    for each bm in bookmarks:

        if bm.ExpiresAt != null && bm.ExpiresAt < Now:
            await store.DeleteBookmarkAsync(bm.BookmarkId)
            continue   // expired — treat as missed event

        // Optimistic lock: tag instance row with ETag before read
        instance = await store.LoadInstanceAsync(bm.InstanceId)
        pointer  = instance.Pointers.Get(bm.PointerId)

        if pointer.Status != Suspended:
            continue   // race condition — another event already resumed this

        // Inject event payload into state so the node can read it
        instance.State.NodeOutputs["__resume_payload"] = DeepClone(event.Payload)

        pointer.Status = Active
        pointer.SuspendedOn = null
        instance.Status = Running

        await store.DeleteBookmarkAndSaveInstanceAsync(bm.BookmarkId, instance)

        // Resume execution — pointer restarts at the node that suspended it
        _ = Task.Run(() =>
            ExecutePointerAsync(instance, pointer, def, triggerCtx: null))

// On process restart — bookmarks still in Redis, instances in SQL
function OnStartup():
    // Reload any Running instances that were mid-execution at crash time
    orphaned = store.GetRunningInstances()
    for each inst in orphaned:
        activePointers = inst.Pointers.Where(p => p.Status == Active)
        for each ptr in activePointers:
            _ = Task.Run(() => ExecutePointerAsync(inst, ptr, def, null))
Error handling + retry policy
function ExecuteWithRetryAsync(
        executor: INodeExecutor,
        ctx:      ExecutionContext,
        policy:   RetryPolicy?) -> NodeResult:

    policy = policy ?? defaultPolicy
    attempt = 0

    loop:
        attempt++
        ctx.Pointer.AttemptCount = attempt

        try:
            using cts = CancellationTokenSource(ctx.Node.Timeout ?? defaultTimeout):
                result = await executor.ExecuteAsync(ctx, cts.Token)
                return result

        catch (OperationCanceledException):
            error = new NodeError("timeout", attempt)
            // timeout counts as an attempt

        catch (ex: Exception):
            error = new NodeError(ex.Message, attempt)

        if attempt >= policy.MaxRetries:
            return NodeResult.Error(error)

        backoff = policy.BackoffMs * (2 ^ (attempt - 1))  // exponential
        jitter  = Random.Next(0, backoff / 4)
        await Task.Delay(backoff + jitter)

function HandleErrorAsync(
        instance: WorkflowInstance,
        pointer:  ExecutionPointer,
        error:    NodeError,
        def:      WorkflowDefinition):

    pointer.Status = Failed
    Log(error, instance.InstanceId, pointer.CurrentNodeId)

    // Check if an error edge exists from this node
    errorEdge = def.Edges.FirstOrDefault(
        e => e.SourceNodeId == pointer.CurrentNodeId
          && e.SourcePort == "error")

    if errorEdge != null:
        // Route to explicit error handler node in graph
        pointer.CurrentNodeId = errorEdge.TargetNodeId
        pointer.Status = Active
        await ExecutePointerAsync(instance, pointer, def, null)
        return

    switch def.ErrorPolicy.OnExhausted:
        case Fail:
            instance.Status = Failed
            await store.SaveInstanceAsync(instance)
            eventBus.Publish(new WorkflowFailedEvent(instance))

        case Compensate:
            await RunCompensationAsync(instance, def)

        case Skip:
            // Mark pointer complete, allow other branches to continue
            pointer.Status = Completed
            CheckWorkflowCompletion(instance)
Compensation — undo already-executed nodes
function RunCompensationAsync(
        instance: WorkflowInstance,
        def:      WorkflowDefinition):

    instance.Status = Compensating
    await store.SaveInstanceAsync(instance)

    // Walk execution log in reverse order
    for each entry in instance.State.ExecutionLog.Reverse():

        node = def.Nodes.Get(entry.NodeId)
        if node == null: continue

        // Only nodes that implement ICompensatable get rolled back
        if node.TypeKey not in compensatableTypes: continue

        using scope = services.CreateScope():
            executor = scope.GetKeyedService<ICompensatableExecutor>(node.TypeKey)

            compensationCtx = new CompensationContext {
                InstanceId: instance.InstanceId
                NodeId:     node.NodeId
                // Original output available to undo the action
                OriginalOutput: instance.State.NodeOutputs[node.NodeId]
                Services:   scope
            }

            try:
                await executor.CompensateAsync(compensationCtx)
            catch (ex):
                // Compensation failure is logged but doesn't stop the rollback chain
                Log("Compensation failed for node {node.NodeId}: {ex}")

    instance.Status = Failed
    await store.SaveInstanceAsync(instance)
    eventBus.Publish(new WorkflowCompensatedEvent(instance))
Workflow within a workflow — sub-workflow node
// A node executor that spawns a child workflow and waits for it
class SubWorkflowExecutor : INodeExecutor {

    function ExecuteAsync(ctx: ExecutionContext) -> NodeResult:

        childDefName = ctx.Config["workflowName"]
        childDef = definitionStore.GetLatestEnabled(childDefName, ctx.WorkspaceId)

        if childDef == null: throw WorkflowNotFound(childDefName)

        // Map parent state into child's initial variables
        inputMapping = ctx.Config["inputMapping"]   // { childVar: parentExpression }
        childInitial = MapVariables(inputMapping, ctx.State)

        childEvent = new SubWorkflowTriggerEvent {
            CorrelationId:    ctx.InstanceId.ToString()
            ParentInstanceId: ctx.InstanceId
            ParentNodeId:     ctx.NodeId
            InitialOverrides: childInitial
        }

        // Engine starts child; child sets ParentInstanceId + ParentNodeId
        childInstance = await engine.StartAsync(childDef, childEvent)

        // Parent pointer suspends on a bookmark keyed to the child instance
        bookmarkName = "subworkflow:completed:{childInstance.InstanceId}"

        return NodeResult.Suspend(new BookmarkRequest {
            Name:    bookmarkName
            Payload: { childInstanceId: childInstance.InstanceId }
        })

// When the child workflow reaches Completed status:
function CheckWorkflowCompletion(instance: WorkflowInstance):

    if instance.Pointers.All(p => p.Status == Completed):
        instance.Status = Completed
        await store.SaveInstanceAsync(instance)

        if instance.ParentInstanceId != null:
            // Collect output variables tagged for export
            outputMapping = parentDef.Nodes.Get(instance.ParentNodeId)
                                .Config["outputMapping"]
            exportedVars = MapVariables(outputMapping, instance.State)

            // Resume parent by publishing to its bookmark
            resumeName = "subworkflow:completed:{instance.InstanceId}"
            await engine.ResumeAsync(new DeviceEvent {
                BookmarkOverride: resumeName
                Payload: exportedVars
            })

        else:
            eventBus.Publish(new WorkflowCompletedEvent(instance))
ExecutionContext — the executor's full API
// Thread-safe, scoped per node execution
class ExecutionContext {

    // Identity
    InstanceId:    Guid
    NodeId:        string
    CorrelationId: string
    WorkspaceId:   Guid

    // Input
    Config:        JsonElement     // deep-cloned node config
    TriggerCtx:    TriggerContext? // original triggering event (only on first node)

    // State access — namespaced reads to avoid key collisions
    function GetVariable<T>(name: string) -> T?
    function SetVariable<T>(name: string, value: T)

    // Read output from a previously executed node
    function GetNodeOutput<T>(nodeId: string) -> T?
        // returns State.NodeOutputs[nodeId] deserialised as T

    // Services (scoped DI — safe for DbContext, HttpClient, etc.)
    Services: IServiceProvider

    // Logging (pre-correlated with instanceId + nodeId)
    Logger: ILogger

    // Cancellation (carries node timeout)
    CancellationToken: CancellationToken
}

// What a node executor returns
record NodeResult {
    ActivatedPort: string          // which outbound port to follow
    Output:        JsonElement?   // written to State.NodeOutputs[NodeId]
    IsSuspend:     bool
    Bookmark:      BookmarkRequest?
    IsError:       bool
    Error:         NodeError?

    // Factory helpers
    static Done(output?)    -> NodeResult  // port = "done"
    static Port(name, out?) -> NodeResult  // port = name (true/false/etc.)
    static Suspend(bm)      -> NodeResult
    static Error(err)        -> NodeResult
}
ConcurrencyPolicy enforcement — full detail
// Stored in WorkflowDefinition. Evaluated at trigger time in RuntimeRegistry.

enum ConcurrencyPolicy {
    AllowParallel    // default — every trigger starts a new instance
    Queue            // only one running; extras queued, drain on completion
    CancelPrevious   // cancel any running instance before starting new
    DropNew          // if one is running, ignore the new trigger entirely
    Debounce         // wait N ms after last trigger before starting
}

// Debounce implementation:
function HandleDebounce(def: WorkflowDefinition, event: DeviceEvent):
    key = "debounce:{def.Id}:{event.CorrelationId}"
    // Redis SET with EX (expiry) = debounce window
    // If key already exists, the previous trigger is replaced
    await redis.SetAsync(key, event, ex: def.DebounceWindow)
    // A background watcher fires StartAsync when the key expires

// Queue drain — called when an instance completes:
function OnInstanceCompleted(instance: WorkflowInstance):
    def = definitionStore.Get(instance.DefinitionId)
    if def.ConcurrencyPolicy != Queue: return

    nextEvent = queue.Dequeue(def.Id)
    if nextEvent != null:
        engine.StartAsync(def, nextEvent)

// Graceful cancel — sets status, deletes bookmarks, publishes event
function CancelAsync(instance: WorkflowInstance, reason: string):
    instance.Status = Cancelled
    await store.DeleteAllBookmarksAsync(instance.InstanceId)
    await store.SaveInstanceAsync(instance)
    eventBus.Publish(new WorkflowCancelledEvent(instance, reason))
